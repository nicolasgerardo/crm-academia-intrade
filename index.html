// ============================================================ // Leads.gs - Motor de gestión de leads para Titanes // ============================================================ // Modelo: // - Cada lead tiene un estado POR CAMPAÑA (no global). // - Los Titanes ven hasta 5 leads asignados a la vez. // - Al cambiar el estado de un lead (sacarlo de "No contactado"), // se libera cupo y el sistema autoasigna otro del pool. // - LockService evita que dos Titanes tomen el mismo lead. // // Hoja 'leads_estados': // lead_email | campania_id | estado | titan_asignado | tipo | // acompanante_de | fecha_asignacion | fecha_ultimo_cambio | notas | // registrado_por // // tipo: 'principal' | 'acompanante' // registrado_por: email de quien creó/registró el lead (para métricas) // ============================================================ var LEAD_MAX_ASIGNADOS = 5; // tope de leads activos por Titán var LEAD_MAX_INTENTOS = 3; // intentos sin respuesta antes de liberar al pool (configurable en Properties) // Lee el máximo de intentos (configurable desde el CRM vía Properties) function leadMaxIntentos() { var v = PropertiesService.getScriptProperties().getProperty('LEAD_MAX_INTENTOS'); return v ? parseInt(v, 10) : LEAD_MAX_INTENTOS; } // Los 12 estados del lead (clave interna : etiqueta visible) var LEAD_ESTADOS = [ { val: 'no_contactado', label: 'No contactado (virgen)' }, { val: 'no_respondio', label: 'No respondió' }, { val: 'confirmado', label: 'Confirmado' }, { val: 'acompanante', label: 'Acompañante' }, { val: 'pendiente_confirmar',label: 'Pendiente por confirmar' }, { val: 'whatsapp_enviado', label: 'WhatsApp enviado' }, { val: 'confirmado_whatsapp',label: 'Confirmado WhatsApp' }, { val: 'vip_llamada', label: 'VIP llamada' }, { val: 'pendiente_vip', label: 'Pendiente VIP' }, { val: 'duplicado', label: 'Duplicado' }, { val: 'erroneo', label: 'Erróneo' }, { val: 'vip_adicional', label: 'VIP adicional' } ]; // Estados que cuentan como "ya trabajado" (liberan cupo de los 5) var LEAD_ESTADOS_CERRADOS = ['confirmado','confirmado_whatsapp','vip_llamada','duplicado','erroneo','vip_adicional','acompanante']; function getHojaLeadsEstados() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('leads_estados'); if (!sheet) { sheet = ss.insertSheet('leads_estados'); sheet.appendRow(['lead_email','campania_id','estado','titan_asignado','tipo','acompanante_de','fecha_asignacion','fecha_ultimo_cambio','notas','registrado_por','intentos']); sheet.setFrozenRows(1); } return sheet; } // ============================================================ // Devuelve la lista de estados (para poblar el frontend) // ============================================================ function crmLeadsEstados() { return { ok: true, data: LEAD_ESTADOS }; } // ============================================================ // MIS LEADS: los 5 (o menos) asignados a un Titán en una campaña. // Si tiene menos de 5 activos, autoasigna del pool hasta completar. // ============================================================ function crmMisLeads(email, campaniaId) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); // espera hasta 10s por el lock (anti-robo) campaniaId = campaniaId || obtenerCampaniaActivaId(); var sheet = getHojaLeadsEstados(); var data = sheet.getDataRange().getValues(); var h = data[0]; var idx = mapHeaders(h); // Índice de la maestra UNA sola vez (evita leer 900 filas por cada lead) var indiceMaestra = construirIndiceMaestra(); // 1. Contar leads activos (no cerrados) ya asignados a este Titán en esta campaña var misActivos = []; for (var i = 1; i < data.length; i++) { if (String(data[i][idx.titan_asignado]).toLowerCase() === email.toLowerCase() && data[i][idx.campania_id] === campaniaId) { var est = data[i][idx.estado]; if (LEAD_ESTADOS_CERRADOS.indexOf(est) < 0) { misActivos.push(filaALead(data[i], idx, i + 1, indiceMaestra)); } } } // 2. Si tiene menos de 5 activos, autoasignar del pool var faltan = LEAD_MAX_ASIGNADOS - misActivos.length; if (faltan > 0) { var nuevos = autoasignarLeads(sheet, data, idx, email, campaniaId, faltan, indiceMaestra); misActivos = misActivos.concat(nuevos); } return { ok: true, data: misActivos, campania_id: campaniaId, max: LEAD_MAX_ASIGNADOS }; } catch (e) { return { ok: false, error: e.message }; } finally { try { lock.releaseLock(); } catch (e) {} } } // Autoasigna leads al Titán: primero los liberados (en el pool, sin titán), // luego contactos nuevos de la campaña. Reparto aleatorio (baraja). function autoasignarLeads(sheetEstados, dataEstados, idx, email, campaniaId, cantidad, indiceMaestra) { indiceMaestra = indiceMaestra || construirIndiceMaestra(); var nuevosLeads = []; // --- PASO 1: tomar leads LIBERADOS (registro existente, misma campaña, sin titán, virgen) --- var liberadosFilas = []; for (var i = 1; i < dataEstados.length; i++) { if (dataEstados[i][idx.campania_id] === campaniaId && !String(dataEstados[i][idx.titan_asignado] || '').trim() && dataEstados[i][idx.estado] === 'no_contactado') { liberadosFilas.push(i + 1); } } // Barajar aleatoriamente los liberados liberadosFilas = barajar(liberadosFilas); for (var L = 0; L < liberadosFilas.length && nuevosLeads.length < cantidad; L++) { var fila = liberadosFilas[L]; sheetEstados.getRange(fila, idx.titan_asignado + 1).setValue(email); sheetEstados.getRange(fila, idx.fecha_asignacion + 1).setValue(new Date()); // Usar la fila ya cargada en memoria (no releer) var rowVals = dataEstados[fila - 1].slice(); rowVals[idx.titan_asignado] = email; nuevosLeads.push(filaALead(rowVals, idx, fila, indiceMaestra)); } // --- PASO 2: si aún faltan, asignar contactos nuevos del pool de la maestra --- var faltan = cantidad - nuevosLeads.length; if (faltan <= 0) return nuevosLeads; var ss = SpreadsheetApp.getActiveSpreadsheet(); var maestra = ss.getSheetByName('contactos_maestra'); if (!maestra) return nuevosLeads; var mData = maestra.getDataRange().getValues(); var mh = mData[0]; var mIdxEmail = mh.indexOf('email'); var mIdxNombre = mh.indexOf('nombre'); var mIdxApellido = mh.indexOf('apellido'); var mIdxTel = mh.indexOf('telefono'); var mIdxWa = mh.indexOf('whatsapp'); var mIdxGrupo = mh.indexOf('grupo_evento'); var mIdxFuente = mh.indexOf('fuente_primaria'); var mIdxEtq = mh.indexOf('etiquetas'); // Emails que YA tienen registro en esta campaña (no duplicar) var yaEnCampania = {}; for (var e = 1; e < dataEstados.length; e++) { if (dataEstados[e][idx.campania_id] === campaniaId) { yaEnCampania[String(dataEstados[e][idx.lead_email]).toLowerCase()] = true; } } // Recolectar candidatos del pool (Money Magnet, sin registro en campaña) var candidatos = []; for (var j = 1; j < mData.length; j++) { var em = String(mData[j][mIdxEmail] || '').trim().toLowerCase(); if (!em || em.indexOf('@') < 0) continue; if (yaEnCampania[em]) continue; var fuente = String(mData[j][mIdxFuente] || '').toLowerCase(); var etq = String(mData[j][mIdxEtq] || '').toLowerCase(); var esMM = fuente.indexOf('money_magnet') >= 0 || fuente.indexOf('landing') >= 0 || etq.indexOf('money_magnet') >= 0; if (!esMM) continue; candidatos.push(j); } // Barajar candidatos (reparto aleatorio) candidatos = barajar(candidatos); var nuevasFilas = []; for (var c = 0; c < candidatos.length && nuevasFilas.length < faltan; c++) { var jj = candidatos[c]; var em2 = String(mData[jj][mIdxEmail]).trim().toLowerCase(); var ahora = new Date(); nuevasFilas.push([em2, campaniaId, 'no_contactado', email, 'principal', '', ahora, ahora, '', '', 0]); nuevosLeads.push({ lead_email: em2, nombre: mIdxNombre >= 0 ? (mData[jj][mIdxNombre] || '') : '', apellido: mIdxApellido >= 0 ? (mData[jj][mIdxApellido] || '') : '', telefono: mIdxTel >= 0 ? (mData[jj][mIdxTel] || '') : '', whatsapp: mIdxWa >= 0 ? (mData[jj][mIdxWa] || '') : '', grupo: mIdxGrupo >= 0 ? (mData[jj][mIdxGrupo] || '') : '', estado: 'no_contactado', tipo: 'principal', acompanante_de: '', intentos: 0, campania_id: campaniaId }); } if (nuevasFilas.length > 0) { sheetEstados.getRange(sheetEstados.getLastRow() + 1, 1, nuevasFilas.length, nuevasFilas[0].length).setValues(nuevasFilas); } return nuevosLeads; } // Baraja un array (Fisher-Yates) para reparto aleatorio function barajar(arr) { for (var i = arr.length - 1; i > 0; i--) { var k = Math.floor(Math.random() * (i + 1)); var tmp = arr[i]; arr[i] = arr[k]; arr[k] = tmp; } return arr; } // ============================================================ // CAMBIAR ESTADO de un lead (libera cupo si pasa a cerrado) // ============================================================ function crmCambiarEstadoLead(email, campaniaId, leadEmail, nuevoEstado, opts) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); opts = opts || {}; campaniaId = campaniaId || obtenerCampaniaActivaId(); var sheet = getHojaLeadsEstados(); var data = sheet.getDataRange().getValues(); var h = data[0]; var idx = mapHeaders(h); for (var i = 1; i < data.length; i++) { if (String(data[i][idx.lead_email]).toLowerCase() === String(leadEmail).toLowerCase() && data[i][idx.campania_id] === campaniaId) { // Verificar que el lead sea de este Titán (o admin) var esAdmin = rolDeEmail(email) === 'admin'; if (!esAdmin && String(data[i][idx.titan_asignado]).toLowerCase() !== email.toLowerCase()) { return { ok: false, error: 'Este lead no está asignado a ti' }; } sheet.getRange(i + 1, idx.fecha_ultimo_cambio + 1).setValue(new Date()); if (opts.notas != null) sheet.getRange(i + 1, idx.notas + 1).setValue(opts.notas); // Estado "no respondió": incrementar contador de intentos if (nuevoEstado === 'no_respondio') { var intentosActuales = Number(data[i][idx.intentos] || 0) + 1; var maxIntentos = leadMaxIntentos(); if (intentosActuales >= maxIntentos) { // Liberar al pool: quitar asignación y volver a virgen para que otro lo tome sheet.getRange(i + 1, idx.estado + 1).setValue('no_contactado'); sheet.getRange(i + 1, idx.titan_asignado + 1).setValue(''); sheet.getRange(i + 1, idx.intentos + 1).setValue(0); sheet.getRange(i + 1, idx.notas + 1).setValue( (opts.notas ? opts.notas + ' | ' : '') + 'Liberado al pool tras ' + maxIntentos + ' intentos' ); return { ok: true, liberado: true, mensaje: 'Lead liberado al pool tras ' + maxIntentos + ' intentos sin respuesta' }; } else { sheet.getRange(i + 1, idx.estado + 1).setValue('no_respondio'); sheet.getRange(i + 1, idx.intentos + 1).setValue(intentosActuales); return { ok: true, intentos: intentosActuales, restantes: maxIntentos - intentosActuales }; } } sheet.getRange(i + 1, idx.estado + 1).setValue(nuevoEstado); if (nuevoEstado === 'acompanante') { sheet.getRange(i + 1, idx.tipo + 1).setValue('acompanante'); if (opts.acompanante_de) sheet.getRange(i + 1, idx.acompanante_de + 1).setValue(opts.acompanante_de); } return { ok: true }; } } return { ok: false, error: 'Lead no encontrado en esta campaña' }; } catch (e) { return { ok: false, error: e.message }; } finally { try { lock.releaseLock(); } catch (e) {} } } // ============================================================ // Helpers // ============================================================ function mapHeaders(h) { return { lead_email: h.indexOf('lead_email'), campania_id: h.indexOf('campania_id'), estado: h.indexOf('estado'), titan_asignado: h.indexOf('titan_asignado'), tipo: h.indexOf('tipo'), acompanante_de: h.indexOf('acompanante_de'), fecha_asignacion: h.indexOf('fecha_asignacion'), fecha_ultimo_cambio: h.indexOf('fecha_ultimo_cambio'), notas: h.indexOf('notas'), registrado_por: h.indexOf('registrado_por'), intentos: h.indexOf('intentos') }; } // Construye un índice {email: {datos}} de la maestra UNA sola vez function construirIndiceMaestra() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var maestra = ss.getSheetByName('contactos_maestra'); var indice = {}; if (!maestra) return indice; var mData = maestra.getDataRange().getValues(); var mh = mData[0]; var iE = mh.indexOf('email'); var iN = mh.indexOf('nombre'), iA = mh.indexOf('apellido'); var iT = mh.indexOf('telefono'), iW = mh.indexOf('whatsapp'), iG = mh.indexOf('grupo_evento'); for (var k = 1; k < mData.length; k++) { var em = String(mData[k][iE] || '').toLowerCase(); if (!em) continue; indice[em] = { nombre: iN>=0 ? (mData[k][iN]||'') : '', apellido: iA>=0 ? (mData[k][iA]||'') : '', telefono: iT>=0 ? (mData[k][iT]||'') : '', whatsapp: iW>=0 ? (mData[k][iW]||'') : '', grupo: iG>=0 ? (mData[k][iG]||'') : '' }; } return indice; } function filaALead(row, idx, filaNum, indiceMaestra) { var em = String(row[idx.lead_email]).toLowerCase(); var info = (indiceMaestra && indiceMaestra[em]) || { nombre:'', apellido:'', telefono:'', whatsapp:'', grupo:'' }; return { lead_email: row[idx.lead_email], nombre: info.nombre, apellido: info.apellido, telefono: info.telefono, whatsapp: info.whatsapp, grupo: info.grupo, estado: row[idx.estado], tipo: row[idx.tipo] || 'principal', acompanante_de: row[idx.acompanante_de] || '', notas: row[idx.notas] || '', intentos: Number(row[idx.intentos] || 0), campania_id: row[idx.campania_id] }; } // Devuelve el id de la campaña activa (evento activo) function obtenerCampaniaActivaId() { var props = PropertiesService.getScriptProperties(); return props.getProperty('EVENTO_ACTIVO_ID') || 'money_magnet'; } // ============================================================ // Configurar el máximo de intentos sin respuesta (solo admin) // ============================================================ function crmConfigIntentos(maxIntentos) { try { var n = parseInt(maxIntentos, 10); if (!n || n < 1 || n > 10) return { ok: false, error: 'Debe ser un número entre 1 y 10' }; PropertiesService.getScriptProperties().setProperty('LEAD_MAX_INTENTOS', String(n)); return { ok: true, max_intentos: n }; } catch (e) { return { ok: false, error: e.message }; } } function crmObtenerConfigIntentos() { return { ok: true, max_intentos: leadMaxIntentos() }; }
